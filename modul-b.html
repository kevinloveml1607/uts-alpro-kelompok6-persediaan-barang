Modul B

Penanggung jawab: Lucy (NIM 322610008)

Algoritma Deskriptif

  1. Mulai.
  2. Siapkan variabel rekap: jumlahAman, jumlahMenipis, jumlahHabis, totalTerjual, totalUnitPesan, dan totalBiayaPesan bernilai 0; penjualanTertinggi bernilai -1; kodeTertinggi dan namaTertinggi dikosongkan; lanjut bernilai 'y'.
  3. Selama lanjut bernilai 'y' atau 'Y', ulangi langkah 4 sampai 12.
  4. Baca kodeBarang dan namaBarang.
  5. Baca barangTerjual. Jika bukan angka atau negatif, tampilkan pesan kesalahan dan baca ulang sampai valid.
  6. Baca statusPilihan (1 = Aman, 2 = Menipis, 3 = Habis). Jika di luar rentang 1 sampai 3, tampilkan pesan kesalahan dan baca ulang sampai valid.
  7. Baca jumlahPesan. Jika bukan angka atau negatif, tampilkan pesan kesalahan dan baca ulang sampai valid.
  8. Baca biayaPesan. Jika bukan angka atau negatif, tampilkan pesan kesalahan dan baca ulang sampai valid.
  9. Jika statusPilihan = 1, tambah jumlahAman 1; jika = 2, tambah jumlahMenipis 1; selain itu tambah jumlahHabis 1.
  10. Tambahkan barangTerjual ke totalTerjual, jumlahPesan ke totalUnitPesan, dan biayaPesan ke totalBiayaPesan.
  11. Jika barangTerjual > penjualanTertinggi, perbarui penjualanTertinggi, kodeTertinggi, dan namaTertinggi dengan data barang saat ini.
  12. Tanyakan "Proses barang lain? (y/n)" dan baca lanjut; tolak jawaban selain y, Y, n, N.
  13. Kembali ke langkah 3.
  14. Setelah perulangan berhenti, tampilkan rekapitulasi: jumlah barang aman, menipis, habis, total unit terjual, total unit dipesan, total biaya pemesanan, dan barang dengan penjualan tertinggi.
  15. Selesai.

Source Code dan Penjelasan

  • Variabel rekap dideklarasikan dan diisi nilai awal sebelum perulangan. penjualanTertinggi diisi -1 agar barang pertama, termasuk yang terjual 0, pasti menjadi rekor awal.
  • Perulangan utama while berjalan selama lanjut bernilai 'y' atau 'Y', sehingga banyak barang dapat diproses.
  • Validasi input memakai do-while. Jika input bukan angka (misalnya "ABC"), cin.clear() dan cin.ignore() membersihkan input yang salah lalu nilainya diberi tanda tidak valid, sehingga program menampilkan pesan kesalahan dan meminta input ulang tanpa berhenti.
  • Percabangan status memakai if - else if - else untuk menambah jumlahAman, jumlahMenipis, atau jumlahHabis.
  • Akumulasi memakai operator += untuk total terjual, total unit pesan, dan total biaya pesan.
  • Penjualan tertinggi dicari dengan membandingkan barangTerjual dengan rekor sementara; jika lebih besar, rekor dan identitas barangnya diperbarui.
  • Pertanyaan lanjut memakai do-while yang menolak jawaban selain y, Y, n, dan N.
  • Rekapitulasi ditampilkan setelah perulangan utama berhenti.
#include <iostream>
#include <string>
using namespace std;

int main() {
    // Deklarasi variabel input per barang
    string kodeBarang, namaBarang;
    int barangTerjual = 0, statusPilihan = 0;
    int jumlahPesan = 0, biayaPesan = 0;
    char lanjut = 'y';

    // Deklarasi variabel rekapitulasi (nilai awal sebelum perulangan)
    int jumlahAman = 0, jumlahMenipis = 0, jumlahHabis = 0;
    int totalTerjual = 0, totalUnitPesan = 0, totalBiayaPesan = 0;
    int penjualanTertinggi = -1;   // -1 agar barang pertama selalu menjadi rekor awal
    string kodeTertinggi = "", namaTertinggi = "";

    cout << "=== REKAPITULASI PERSEDIAAN BARANG (MODUL B) ===" << endl;

    // Perulangan utama: proses barang selama petugas memilih lanjut
    while (lanjut == 'y' || lanjut == 'Y') {
        cout << "\n--- Hasil Evaluasi Barang ---" << endl;
        cout << "Kode barang : ";
        getline(cin >> ws, kodeBarang);
        cout << "Nama barang : ";
        getline(cin >> ws, namaBarang);

        // Validasi barang terjual (tidak boleh negatif atau bukan angka)
        do {
            cout << "Barang terjual : ";
            if (!(cin >> barangTerjual)) {
                cin.clear();
                cin.ignore(10000, '\n');
                barangTerjual = -1;
            }
            if (barangTerjual < 0)
                cout << "  [!] Harus berupa angka dan tidak boleh negatif." << endl;
        } while (barangTerjual < 0);

        // Validasi status barang (hanya 1, 2, atau 3)
        do {
            cout << "Status (1=Aman, 2=Menipis, 3=Habis): ";
            if (!(cin >> statusPilihan)) {
                cin.clear();
                cin.ignore(10000, '\n');
                statusPilihan = 0;
            }
            if (statusPilihan < 1 || statusPilihan > 3)
                cout << "  [!] Pilih 1, 2, atau 3." << endl;
        } while (statusPilihan < 1 || statusPilihan > 3);

        // Validasi jumlah pesan
        do {
            cout << "Jumlah dipesan : ";
            if (!(cin >> jumlahPesan)) {
                cin.clear();
                cin.ignore(10000, '\n');
                jumlahPesan = -1;
            }
            if (jumlahPesan < 0)
                cout << "  [!] Harus berupa angka dan tidak boleh negatif." << endl;
        } while (jumlahPesan < 0);

        // Validasi biaya pemesanan
        do {
            cout << "Biaya pemesanan : Rp";
            if (!(cin >> biayaPesan)) {
                cin.clear();
                cin.ignore(10000, '\n');
                biayaPesan = -1;
            }
            if (biayaPesan < 0)
                cout << "  [!] Harus berupa angka dan tidak boleh negatif." << endl;
        } while (biayaPesan < 0);

        // Hitung jumlah barang per status
        if (statusPilihan == 1) {
            jumlahAman++;
        } else if (statusPilihan == 2) {
            jumlahMenipis++;
        } else {
            jumlahHabis++;
        }

        // Akumulasi total
        totalTerjual    += barangTerjual;
        totalUnitPesan  += jumlahPesan;
        totalBiayaPesan += biayaPesan;

        // Cari barang dengan penjualan tertinggi
        if (barangTerjual > penjualanTertinggi) {
            penjualanTertinggi = barangTerjual;
            kodeTertinggi = kodeBarang;
            namaTertinggi = namaBarang;
        }

        // Tanya lanjut, tolak jawaban selain y/n
        do {
            cout << "\nProses barang lain? (y/n): ";
            cin >> lanjut;
            cin.ignore(10000, '\n');
            if (lanjut != 'y' && lanjut != 'Y' && lanjut != 'n' && lanjut != 'N')
                cout << "  [!] Jawab y atau n." << endl;
        } while (lanjut != 'y' && lanjut != 'Y' && lanjut != 'n' && lanjut != 'N');
    }

    // Tampilkan rekapitulasi
    cout << "\n--- Rekapitulasi Persediaan ---" << endl;
    cout << "Barang aman         : " << jumlahAman << endl;
    cout << "Barang menipis      : " << jumlahMenipis << endl;
    cout << "Barang habis        : " << jumlahHabis << endl;
    cout << "Total unit terjual  : " << totalTerjual << endl;
    cout << "Total unit dipesan  : " << totalUnitPesan << " unit" << endl;
    cout << "Total biaya pesan   : Rp" << totalBiayaPesan << endl;
    cout << "Penjualan tertinggi : " << kodeTertinggi << " / " << namaTertinggi
         << " (" << penjualanTertinggi << " unit)" << endl;

    cout << "\nProgram selesai." << endl;
    return 0;
}
